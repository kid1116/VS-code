if((0,c.isWindows)()&&/^cl(?:\.exe)?$/i.test(f.basename(this.compiler))){
    const buildScript=f.join(process.env.APPDATA||"","Code","User","cpp-unicode.ps1");
    if(!(0,d.existsSync)(buildScript)){
        l.Notification.showErrorMessage("MSVC build helper not found: "+buildScript);
        return;
    }
    const extraCompilerArgs=[...(0,u.splitArgs)(this.inputFlags),
        ...[...i.Configuration.additionalIncludePaths(),...i.Configuration.includePathsFromCppProperties()]
            .filter(value=>value.trim().length>0).map(value=>"/I"+value)];
    const extraLinkerArgs=(0,u.splitArgs)(this.linkerFlags);
    const encodeArgs=values=>Buffer.from(JSON.stringify(values),"utf8").toString("base64");
    return await new Promise(resolve=>{
        const child=h.spawn("powershell.exe",["-NoLogo","-NoProfile","-NonInteractive","-ExecutionPolicy","Bypass",
            "-File",buildScript,"-Source",this.file.path,"-OutputDirectory",t,
            "-CompilerArgumentsBase64",encodeArgs(extraCompilerArgs),
            "-LinkerArgumentsBase64",encodeArgs(extraLinkerArgs)],
            {cwd:this.file.directory,env:process.env,windowsHide:true});
        let output="",finished=false;
        child.stdout?.setEncoding("utf8");
        child.stderr?.setEncoding("utf8");
        child.stdout?.on("data",text=>{output+=text});
        child.stderr?.on("data",text=>{output+=text});
        child.on("error",error=>{
            if(finished)return;
            finished=true;
            l.Notification.showErrorMessage("Failed to launch MSVC: "+error.message);
            resolve();
        });
        child.on("close",async code=>{
            if(finished)return;
            finished=true;
            try{
                if(code===0&&(0,d.existsSync)(n)){
                    l.Notification.showInformationMessage("Compilation successful (MSVC).");
                    if(e)await e();
                }else{
                    l.Notification.showErrorMessage("Compilation failed (MSVC, exit code "+code+").\n"+output.trim());
                }
            }catch(error){
                l.Notification.showErrorMessage("Failed to run the compiled program: "+error.message);
            }finally{resolve()}
        });
    });
}
